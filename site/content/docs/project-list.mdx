import { CardGrid, LinkCard } from '@astrojs/starlight/components';
import { REPO_URL, groupByCategory, readProjects } from '../../catalog.mjs';

{/* Generated from the project folders themselves — a new folder is a new entry
    here, and a project re-filed in its README moves category on its own. A
    project is shown as a card once it has a guide; until then it is a link to
    its folder, which already holds the CAD, the firmware and the notes. */}

export const showDrafts = import.meta.env.MODE !== 'production';
export const projects = readProjects().map((p) => ({ ...p, guide: p.guide && (showDrafts || !p.draft) }));
export const groups = groupByCategory(projects);
export const written = projects.filter((p) => p.guide).length;

All {projects.length} of them. Every project lives in its own folder in the
<a href={`${REPO_URL}/tree/main/projects`}>GitHub repo</a> — the write-up, the photos, the print
files, the CAD source and the firmware, all in one place — and every one has a page here with
its parts in a 3D viewer and its files ready to download. Some are full build guides; some are
a paragraph and the files, which is the honest length for a bracket.

{groups.map((group) => (
	<section>
		<h2 id={group.id}>{group.label}</h2>
		<p>{group.blurb}</p>
		{group.projects.some((p) => p.guide) && (
			<CardGrid>
				{group.projects.filter((p) => p.guide).map((p) => (
					<LinkCard title={p.title} href={`/projects/${p.slug}/`} description={p.summary} />
				))}
			</CardGrid>
		)}
		{group.projects.some((p) => !p.guide && p.files) && (
			<p>
				<strong>Files only, no guide yet:</strong>{' '}
				{group.projects.filter((p) => !p.guide && p.files).map((p, i, all) => (
					<>
						<a href={`${REPO_URL}/tree/main/projects/${p.slug}`}>{p.title}</a>
						{i < all.length - 1 ? ', ' : '.'}
					</>
				))}
			</p>
		)}
		{group.projects.some((p) => !p.guide && !p.files) && (
			<p>
				<strong>Write-up pending:</strong>{' '}
				{group.projects.filter((p) => !p.guide && !p.files).map((p, i, all) => (
					<>
						<a href={`${REPO_URL}/tree/main/projects/${p.slug}`}>{p.title}</a>
						{i < all.length - 1 ? ', ' : '.'}
					</>
				))}
				{' '}Built, not yet written up or published — the folder is a placeholder for now.
			</p>
		)}
	</section>
))}
